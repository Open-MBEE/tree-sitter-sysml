package Definitions {
    abstract part def Vehicle :> Base::Anything {
        attribute mass : MassValue;
    }
    item def Fuel;
    attribute def <mv> MassValue :> ScalarValues::Real;
    enum def Color { red; green; blue; }
    occurrence def Trip;
    port def FuelPort { in item fuel : Fuel; }
    connection def Link { end a : Vehicle; end b : Vehicle; }
    interface def Plug { end p1 : FuelPort; end p2 : ~FuelPort; }
    allocation def Alloc;
    flow def FuelFlow;
    metadata def Safety { attribute level : Integer; }
    action def Drive { in speed : Real; out done : Boolean; }
    calc def Add { in x : Real; in y : Real; return : Real = x + y; }
    state def Machine;
    constraint def Positive { in x : Real; x > 0 }
    requirement def Req { subject s : Vehicle; require constraint { s.mass > 0 } }
    concern def Cost;
    case def Case;
    analysis def Analysis;
    verification def Verify;
    use case def Use;
    view def V;
    viewpoint def VP;
    rendering def R;
    variation part def Choice { variant part a : Vehicle; variant part b : Vehicle; }
    individual part def MyCar :> Vehicle;
    part def Wheeled unions Car, Truck;
    #safe part def Marked;
}
